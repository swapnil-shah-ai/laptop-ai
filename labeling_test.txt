import urllib.request
import json
import time

TICKETS = [
    "My internet has been down for 3 days and nobody is responding to my complaints",
    "How do I update my billing address on the portal?",
    "Thanks for sending the technician so quickly, great service!",
    "I'm being charged for a plan I never subscribed to. This is fraud.",
    "Can you tell me the data limit on my current plan?",
    "Network keeps dropping every evening between 7-9pm for the past week",
    "I want to cancel my connection immediately",
    "Your new app update is really smooth, love the interface",
    "My router is making a buzzing noise and getting very hot",
    "When will 5G be available in my area?"
]

LABELER = "phi3:mini"

def call_ollama(model, prompt):
    data = json.dumps({"model": model, "prompt": prompt, "stream": False, "options": {"temperature": 0}}).encode()
    req = urllib.request.Request("http://localhost:11434/api/generate", data=data, headers={"Content-Type": "application/json"})
    response = urllib.request.urlopen(req, timeout=120)
    return json.loads(response.read())["response"].strip()

def label_ticket(ticket):
    prompt = (
        "You are a data labeling system. Classify customer support tickets.\n\n"
        "Example 1:\n"
        "Ticket: \"I was charged twice for my last bill\"\n"
        "category: billing\n"
        "urgency: high\n"
        "sentiment: negative\n\n"
        "Example 2:\n"
        "Ticket: \"How do I check my data balance?\"\n"
        "category: general\n"
        "urgency: low\n"
        "sentiment: neutral\n\n"
        "Example 3:\n"
        "Ticket: \"Your technician fixed it in 10 minutes, amazing work\"\n"
        "category: feedback\n"
        "urgency: low\n"
        "sentiment: positive\n\n"
        "Example 4:\n"
        "Ticket: \"My connection drops every evening for the past week\"\n"
        "category: network\n"
        "urgency: high\n"
        "sentiment: negative\n\n"
        "Example 5:\n"
        "Ticket: \"The router is overheating and making noise\"\n"
        "category: hardware\n"
        "urgency: high\n"
        "sentiment: negative\n\n"
        "Now classify this ticket. Reply ONLY in the exact format shown above. "
        "Do not add explanations. Do not use markdown. Do not add any text before or after the three lines.\n\n"
        "Ticket: \"" + ticket + "\""
    )
    return call_ollama(LABELER, prompt)

def run():
    print("PHASE 15: DATA LABELING (with prompt engineering)")
    print(f"Labeler: {LABELER} | Tickets: {len(TICKETS)}")
    print("Changes: few-shot examples + temperature 0\n")

    results = []
    for i, ticket in enumerate(TICKETS):
        print(f"Ticket {i+1}: {ticket[:60]}...")
        print(f"  labeling...", end=" ", flush=True)
        start = time.time()
        labels = label_ticket(ticket)
        print(f"({time.time()-start:.1f}s)")

        cat = urg = sent = "?"
        for line in labels.lower().split("\n"):
            if "category:" in line: cat = line.split(":")[-1].strip()
            if "urgency:" in line: urg = line.split(":")[-1].strip()
            if "sentiment:" in line: sent = line.split(":")[-1].strip()

        results.append({"ticket": ticket[:50], "category": cat, "urgency": urg, "sentiment": sent})
        print(f"  > {cat} | {urg} | {sent}\n")

    print("=" * 75)
    print("LABELED DATASET")
    print("=" * 75)
    print(f"{'#':<4}{'Ticket':<52}{'Category':<12}{'Urgency':<10}{'Sentiment'}")
    print("-" * 75)
    for i, r in enumerate(results):
        print(f"{i+1:<4}{r['ticket']:<52}{r['category']:<12}{r['urgency']:<10}{r['sentiment']}")

run()
